const express = require("express"); const multer = require("multer"); const cors = require("cors"); const fs = require("fs"); const app = express(); app.use(cors()); const upload = multer({ dest: "uploads/" }); let files = []; app.post("/upload", upload.single("file"), (req, res) => { const file = { id: req.file.filename, name: req.file.originalname }; files.push(file); res.send("Uploaded"); }); app.get("/files", (req, res) => { res.json(files); }); app.get("/video/:id", (req, res) => { res.sendFile(__dirname + "/uploads/" + req.params.id); }); app.get("/download/:id", (req, res) => { res.download(__dirname + "/uploads/" + req.params.id); }); app.delete("/delete/:id", (req, res) => { const id = req.params.id; files = files.filter(f => f.id !== id); fs.unlinkSync("uploads/" + id); res.send("Deleted"); }); app.listen(3000, () => console.log("Server running on 3000"));
